                        array2[i] = star;
                        star += starAdd;
                        j++;
                    }

                        System.out.println(array2[j - 1]);

                        star += "*  "; //star 너비 늘려주기
                    }
                    for (i = 0; i < width; i++) {
                        System.out.println(star); //높이 길이 만큼 루프 반복
                    }

                    }
                    for (i = 0; i < length; i++) { //정사각형이라 너비 길이만큼 높이 반복
                        System.out.println(star);
                    }
                    System.out.println("\n계속 할까요? y/n: ");
                    yn = sc.nextLine();
                    if (yn.equals("n")) {
                        System.out.println("\n도형 그리기 교실을 종료합니다");
                        System.out.println("다음 시간에 또 만나요!");

                        for (int y = -radius; y <= radius; y++) {

                            if (x * x + y * y < radius * radius) { // <=로 하니까 끝에 *이 나와서 맘에 안드는데, 추가로 양 쪽에 별을 붙이고 싶으나 시간이 안됨
                                System.out.print(star);

                            } else {
                                System.out.print(space);
                            }
                        }
                        System.out.println();

                    }

                    //농담 아니고 진짜 이렇게 좌표계 스타일로 쉽게 만들 수 있는지 몰랐음
                    //안 찾아봤으면 개삽질 계속 했을듯...

                    /*

                    for (i = 0; i < x; i++) { //별을 그려줌

                            System.out.print(star);
                        }
