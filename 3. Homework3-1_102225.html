void main() {
    Scanner sc = new Scanner(System.in);
    int a; //단
    int b; //사용자가 요구하는 갯수
    int c; //multiplication = (a * i)
    int i; //increment until b
    int k = 0; //while loop counter
    int j = 0; //sum of 구구단
    boolean flag = true;
    int[] array = new int[10]; //하드 코딩. Save slot은 10개까지만.

    System.out.println("\n안녕하세요, 닉스의 구구단 교실에 오신 것을 환영합니다");

    //반복 구간 시작
    while (flag) {
        System.out.println("\n몇 단부터 시작해볼까요?");
        a = sc.nextInt();
        sc.nextLine();
        System.out.println("좋습니다. " + a + "단을 계산해보겠습니다");
        System.out.println("\n몇까지 출력할까요?");
        b = sc.nextInt();
        sc.nextLine();
        System.out.println(b + "까지 출력하겠습니다:");

        for (i = 1; i <= b; i++) {
            c = a * i;
            System.out.println(a + " x " + i + " = " + c);
        }
        if (b%2 == 0) { //짝수일때 총 합 계산... 분명 이 방법을 바란게 아니라 아래 sum += array[i] 방식을 요구하셨을텐데... 수학적으로 풀어버림 ㅋㅋ 이 방법 먼저 찾고 후의 방법 찾음... 하...
            j = a * (1 + b) * b/2;
            System.out.println("\n이번 계산식의 총 값은 " + j + "입니다");
        }
        else if (b%2 == 1) { //홀수일때 총 합 계산
            j = a * ((2 + b) * (b+1)/2 - (b+1)); // 식과 j가 반대로 되니까 안되어서 한참 헤멤
        System.out.println("\n이번 계산식의 총 값은 " + j + "입니다");
        }


        array[k] = j; //루프당 j의 값을 저장.
        System.out.println("array[" + k + "]"+ "에 " + j + "의 값이 저장되었습니다");
        k++; //루프를 추가
        System.out.println(k + "번 루프를 끝냈습니다");
        System.out.println("\n저장 가능 슬롯이 " + (10-k) + "번 남았습니다"); //하드 코딩 외 방법이 있는지 찾아보려 했으나 쉽지 않음
        System.out.print("\n계속 할까요? (y/n): ");

        String choice;
        choice = sc.nextLine();
        if (choice.equals("n")) { //String은 .equals() 인 것을 까먹고 ==를 해서 엄청 헤멤...
            flag = false;
            System.out.println("\n-----구구단이 끝났습니다-----");
            System.out.println(k + "개의 저장된 슬롯이 있습니다");
            System.out.println("해당 슬롯에 저장된 모든 값을 더하는 중입니다...");
            int sum = 0;
            for (i = 0; i < array.length; i++) {
                sum += array[i]; //얘 인터넷에서 찾아서 슬쩍 카피함... sum = sum + array[i] i는 int 안 쓰고 다시 0으로 고정시켜주고 재사용
            }
            System.out.println("총 합은 >> " + sum + " << 입니다");
            break;
            }
    }
}
