void main() {
    Scanner sc = new Scanner(System.in);
    int a; //단
    int b; //갯수
    int c; //총 출력값
    int i; //increment
    boolean flag = true;

    System.out.println("안녕하세요, 닉스의 구구단 교실에 오신 것을 환영합니다");

    //반복 구간 시작
    while (flag) {
        System.out.println("몇 단부터 시작해볼까요?");
        a = sc.nextInt();
        sc.nextLine();
        System.out.println("좋습니다. " + a + "단을 계산해보겠습니다");
        System.out.println("몇까지 출력할까요?");
        b = sc.nextInt();
        sc.nextLine();
        System.out.println(b + "까지 출력하겠습니다:");

        for (i = 1; i <= b; i++) {
            System.out.println(a + " x " + i + " = " + a * i);
        }

        System.out.print("계속 할까요? (y/n): ");
        String choice;
        choice = sc.nextLine();
        if (choice.equals("n")) { //String은 .equals() 인 것을 까먹고 ==를 해서 엄청 헤멤...
            flag = false;
            System.out.println("\n-----계산이 끝났습니다-----");
            //System.out.println("지금까지 저장된 결과의 총 합은 " + c + "입니다");
            break;
        }
    }
}
